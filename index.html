function salvarCampo(turno, index) { const caminho = `${obterCaminhoFirebase()}/${turno}/${index}`; const atendimento = document.getElementById(`${turno}-atendimento-${index}`).value.trim(); const cliente = document.getElementById(`${turno}-cliente-${index}`).value.trim(); const vendedor = document.getElementById(`${turno}-vendedor-${index}`).value.trim(); const anotadoPor = document.getElementById(`${turno}-anotadoPor-${index}`).value.trim(); const obs = document.getElementById(`${turno}-obs-${index}`).value.trim(); const motorista = document.getElementById(`${turno}-motorista-${index}`).value.trim(); // Se todos os campos de texto estiverem vazios, remove a linha do banco de dados if (!atendimento && !cliente && !vendedor && !anotadoPor && !obs && !motorista) { database.ref(caminho).remove(); return; } let dadosAtuais = dadosCache[turno] && dadosCache[turno][index] ? dadosCache[turno][index] : {}; let dataHora = dadosAtuais.dataHora; if (!dataHora || dataHora === '-') { const agora = new Date(); dataHora = `${agora.toLocaleDateString('pt-BR')} ${agora.toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' })}`; } const payload = { atendimento, cliente, vendedor, anotadoPor, obs, motorista, dataHora: dataHora, ok: dadosAtuais.ok || false }; database.ref(caminho).set(payload); }
